<?php

use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

require 'PHPMailer/src/Exception.php';
require 'PHPMailer/src/PHPMailer.php';
require 'PHPMailer/src/SMTP.php';


if ($_SERVER["REQUEST_METHOD"] == "POST") {

    $name = htmlspecialchars($_POST['name']);
    $email = htmlspecialchars($_POST['email']);
    $phone = htmlspecialchars($_POST['phone']);
    $message = htmlspecialchars($_POST['message']);
    $company = htmlspecialchars($_POST['company']);
    $service = htmlspecialchars($_POST['service']);

    $mail = new PHPMailer(true);

    try {

        // SMTP Settings
        #$mail->isSMTP();
        #$mail->Host = 'smtp.hostinger.com';
        $mail->SMTPDebug = 2;
        $mail->Debugoutput = 'html';
        $mail->SMTPAuth = true;

        // Change these
        $mail->Username = 'info@etigait.in';
        $mail->Password = 'RamBoyapati@03';

        $mail->SMTPSecure = PHPMailer::ENCRYPTION_SMTPS;
        $mail->Port = 465;

        // Sender
        $mail->setFrom('info@etigait.in', 'Etiga IT Solutions');

        // Receiver
        $mail->addAddress('info@etigait.in');

        // Reply to visitor
        $mail->addReplyTo($email, $name);

        // Email Content
        $mail->isHTML(true);
        $mail->Subject = "New Contact Form Submission - Etiga IT Solutions";

        $mail->Body = "
<h2>New Contact Form Submission</h2>

<p><strong>Name:</strong> {$name}</p>

<p><strong>Company:</strong> {$company}</p>

<p><strong>Email:</strong> {$email}</p>

<p><strong>Phone:</strong> {$phone}</p>

<p><strong>Service:</strong> {$service}</p>

<p><strong>Message:</strong><br>{$message}</p>
";

        $mail->send();

      echo "<script>
            alert('Thank you! Your message has been sent successfully.');
            window.location.href='index.html#contact';
            </script>";

    } catch (Exception $e) {

        echo "<script>
            alert('Message could not be sent. Error: {$mail->ErrorInfo}');
            window.history.back();
        </script>";

    }
}
?>
